Do not nitpick style or punctuation. - If the message is mostly English, gently encourage trying in ${settings.language} and give the ${settings.language} version as one correction. - "corrections": each "original" must be an EXACT substring of the student's latest message. - "why": a ${correctionLength} explanation in English. - "reply_en": English translation of your reply. - "praise": one short English sentence naming something they did well. - "score": 0–100 accuracy of the latest message. - "suggestions": 2 short possible student replies in ${settings.language}. - For the opening turn, provide a friendly greeting and a simple question, with no corrections and score 100. Return a JSON object with exactly this shape: {"reply":"","reply_en":"","corrections":[{"original":"","corrected":"","why":"","type":"grammar|vocabulary|spelling|word order|other"}],"praise":"","score":100,"suggestions":["",""]}`; } function isValidSettings(body) { const languages = [ "Spanish", "French", "German", "Japanese", "English" ]; const levels = ["A1", "A2", "B1", "B2"]; const topics = [ "Ordering at a café", "Introducing yourself", "Asking for directions", "Talking about school", "Planning a weekend", "Free chat" ]; return body && languages.includes(body.language) && levels.includes(body.level) && topics.includes(body.topic) && ["brief", "full"].includes(body.correctionStyle) && Array.isArray(body.messages) && body.messages.length <= 22; } const SUPPORTED_TRANSLATION_LANGUAGES = [ "Spanish", "French", "German", "Japanese", "English" ]; async function callGemini(env, contents, systemInstruction, maxOutputTokens = 900) { const model = env.GEMINI_MODEL || "gemini-2.5-flash-lite"; const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(model)}:generateContent`; return fetch(endpoint, { method: "POST", headers: { "content-type": "application/json", "x-goog-api-key": env.GEMINI_API_KEY }, body: JSON.stringify({ systemInstruction: { parts: [{ text: systemInstruction }] }, contents, generationConfig: { temperature: 0.3, maxOutputTokens, responseMimeType: "application/json" } }) }); } function extractGeminiText(data) { return data?.candidates?.[0]?.content?.parts ?.map(part => typeof part.text === "string" ? part.text : "") .join("") || ""; } export default { async fetch(request, env) { const { origin, allowed } = validOrigin(request, env); if (!allowed) { return new Response("Origin not allowed", { status: 403 }); } const cors = corsHeaders(origin); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } const url = new URL(request.url); if (request.method === "GET" && url.pathname === "/health") { return json({ ok: true, service: "TalkPartner", provider: "Gemini", version: 2, translation: true }, 200, origin); } if (request.method !== "POST" || (url.pathname !== "/chat" && url.pathname !== "/translate")) { return json({ error: "Not found" }, 404, origin); } if (!env.GEMINI_API_KEY) { return json({ error: "The Gemini service is not configured yet." }, 503, origin); } let body; try { const raw = await request.text(); if (raw.length > 30000) { return json({ error: "Message is too long." }, 413, origin); } body = JSON.parse(raw); } catch { return json({ error: "Invalid request." }, 400, origin); } // Translate a tutor reply into the student's chosen language. if (url.pathname === "/translate") { if ( !body || typeof body.text !== "string" || !body.text.trim() || body.text.length > 3000 || !SUPPORTED_TRANSLATION_LANGUAGES.includes(body.targetLanguage) ) { return json({ error: "Please provide a valid tutor reply and translation language." }, 400, origin); } try { const upstream = await callGemini( env, [{ role: "user", parts: [{ text: `Translate the following text into ${body.targetLanguage}:\n\n${body.text}` }] }], `You are a careful translator. [{ role: "user", parts: [{ text: "Start the conversation now." }] }] : messages; try { const upstream = await callGemini( env, apiMessages, tutorPrompt(body), 900 ); let data; try { data = await upstream.json(); } catch { return json({ error: "The AI provider returned an unreadable response." }, 502, origin); } if (!upstream.ok) { const status = upstream.status === 429 ? 429 : 502; return json({ error: upstream.status === 429 ? "The free Gemini usage limit has been reached for now. Please try again later." : "The Gemini service could not complete the request. Please try again." }, status, origin); } let result; try { result = JSON.parse(extractGeminiText(data)); } catch { return json({ error: "The tutor returned an unreadable response. Please try again." }, 502, origin); } } };
